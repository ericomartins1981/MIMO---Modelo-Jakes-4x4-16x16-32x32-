import matplotlib.pyplot as plt
import numpy as np
from scipy.linalg import sqrtm
from scipy.special import j0

plt.rcParams['font.family'] = 'serif'

def matriz_correlacao_jakes(n_antenas, coef):
  if coef == 0.0:
    return np.eye(n_antenas, dtype=float)
  indices = np.arange(n_antenas)
  distancia = np.abs(indices[:, None] - indices[None, :])
  R = j0(2.0 * np.pi * coef * distancia)
  autovalores, autovetores = np.linalg.eigh(R)
  autovalores = np.maximum(autovalores, 0.0)
  return (autovalores * autovetores) @ autovetores.T

def capacidade_ergodica_mimo_jakes(
    snr_db_array, lt=4, lr=4, coef_t=0.0, coef_r=0.0, n_amostras=100000, seed=42
):
  rng = np.random.default_rng(seed)
  snr_lin = 10 ** (snr_db_array / 10.0)
  Rt = matriz_correlacao_jakes(lt, coef_t)
  Rr = matriz_correlacao_jakes(lr, coef_r)
  Rt_sqrt = np.real(sqrtm(Rt))
  Rr_sqrt = np.real(sqrtm(Rr))
  real_part = rng.standard_normal((n_amostras, lr, lt))
  imag_part = rng.standard_normal((n_amostras, lr, lt))
  Gw = (real_part + 1j * imag_part) / np.sqrt(2.0)
  G = np.matmul(Rr_sqrt, np.matmul(Gw, Rt_sqrt))
  G_GH = np.matmul(G, np.conj(np.transpose(G, (0, 2, 1))))
  autovalores = np.linalg.eigvalsh(G_GH)
  autovalores = np.maximum(autovalores, 0.0)
  capacidades = np.empty_like(snr_lin, dtype=float)
  for idx, rho in enumerate(snr_lin):
    escala = rho / lt
    log2_det_amostras = np.sum(np.log2(1.0 + escala * autovalores), axis=1)
    capacidades[idx] = np.mean(log2_det_amostras)
  return capacidades

snr_db = np.linspace(-5, 25, 121)

cenarios = [
    (0.0, 0.0, 'Descorrelacionado ($\\rho_t=0, \\rho_r=0$)', 'tab:blue', '-'),
    (
        0.0,
        0.95,
        'Receptor correlacionado ($\\rho_t=0, \\rho_r=0.95$)',
        'tab:orange',
        '--',
    ),
    (
        0.95,
        0.95,
        'Ambos correlacionados ($\\rho_t=0.95, \\rho_r=0.95$)',
        'tab:red',
        '-.',
    ),
]

plt.figure(figsize=(10, 6), dpi=120)

for coef_t, coef_r, rotulo, cor, estilo in cenarios:
  c_erg = capacidade_ergodica_mimo_jakes(
      snr_db, lt=4, lr=4, coef_t=coef_t, coef_r=coef_r, n_amostras=100000
  )
  plt.plot(
      snr_db, c_erg, label=rotulo, color=cor, linestyle=estilo, linewidth=2.4
  )

plt.xlim(-5, 25)
plt.ylim(0, None)
plt.xlabel('SNR (dB)', fontsize=13)
plt.ylabel('Capacidade ergódica (bits/s/Hz)', fontsize=13)
plt.title(
    'Capacidade ergódica MIMO ($4 \\times 4$) — Modelo de Jakes ($ \\rho_c= 0.95$)',
    fontsize=14,
)
plt.grid(True, linestyle='--', linewidth=0.75, alpha=0.7)
plt.legend(loc='upper left', fontsize=11)
plt.tight_layout()
plt.show()
